-- 037 P3 evidence driver, CORE side: prints what the harness's own
-- `result_env_model_content` (src/core/phase_vocab.ail) builds for the
-- envelopes the skills package's `envelope_message` is tested on. Run from the
-- worktree root; its output is compared with the package side's.
module scratch/envelope_parity_core

import std/io (println)
import std/crypto (sha256Hex)
import std/json (jo, kv, js)
import std/string (concat, intToStr, length, repeat)
import src/core/tool_contract (ToolResultEnvelope)
import src/core/phase_vocab (result_env_model_content)

func skill_env(call_id: string, dir_name: string, text: string) -> ToolResultEnvelope {
  { tool_call_id: call_id, tool: "Skill", exit_code: 0, stdout: concat([".motoko/skills/", dir_name, "\n", text]), stderr: "", metadata: jo([]) }
}

func small() -> ToolResultEnvelope {
  skill_env("call_0123456789abcdef0123456789abcdef", "observer", "---\nname: observer\ndescription: \"Say \\\"hi\\\"\"\n---\n# Body <&> \u{00E9}\n")
}

func tricky_text() -> string {
  concat(["---\nname: a\ndescription: d\n---\n", repeat("tab\there \"quoted\" back\\slash <tag> & amp \u{00E9}\u{2028}\u{0001}\r\n", 800)])
}

export func main() -> () ! {IO} {
  let a = result_env_model_content(small());
  let b = result_env_model_content({ small() | metadata: jo([kv("cells", js("c")), kv("kept", js("k")), kv("images", js("i")), kv("jsonOutputs", js("j"))]) });
  let c = result_env_model_content(skill_env("call_00000000000000000000000000000000", "a", tricky_text()));
  let d = result_env_model_content(skill_env("call_0123456789abcdef0123456789abcdef", "workdir-stamp", concat(["---\nname: workdir-stamp\ndescription: x\n---\n", repeat("0123456789", 6100)])));
  println(concat(["A ", a]));
  println(concat(["B ", b]));
  println(concat(["C chars=", intToStr(length(c)), " sha256=", sha256Hex(c)]));
  println(concat(["D chars=", intToStr(length(d)), " sha256=", sha256Hex(d)]))
}
